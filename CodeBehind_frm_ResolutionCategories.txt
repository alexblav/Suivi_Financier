' =====================================================================================
' Ce code gère deux événements automatiques de la feuille :
'   - Worksheet_Change  : se déclenche dès qu'une cellule de la feuille change de
'                         valeur (ex. : l'utilisateur choisit une catégorie dans une
'                         liste déroulante). On y détecte le choix et on met à jour
'                         le statut + le compteur automatiquement (comportement
'                         "Option B" validé avec toi).
'   - Worksheet_Deactivate : se déclenche dès que l'utilisateur essaie de quitter
'                         cette feuille (changer d'onglet, Ctrl+PageUp/Down...).
'                         Tant que la saisie est en cours, on le ramène de force
'                         sur cette feuille : c'est le "verrou modal" convenu.
' =====================================================================================

Private Sub Worksheet_Change(ByVal Target As Range)

    ' --- Garde-fou n°1 : ignorer les changements que NOUS provoquons nous-mêmes
    ' (remplissage automatique du tableau par RemplirTableauCas) ---
    If g_ChargementEnCours Then Exit Sub

    ' --- Garde-fou n°2 : ne réagir que si la cellule modifiée est dans la colonne
    ' "Categorie" ET dans la plage des lignes de données du tableau ---
    If Target.Column <> Me.Range(COL_CATEGORIE & "1").Column Then Exit Sub
    If Target.Row < LIGNE_PREMIERE_DONNEE Then Exit Sub
    If Target.Row > LIGNE_PREMIERE_DONNEE + g_NbCasAmbigus - 1 Then Exit Sub

    ' --- Garde-fou n°3 : si plusieurs cellules ont changé en une fois (ex. : collage
    ' multiple), on ne traite que la première pour rester simple et prévisible ---
    If Target.Cells.count > 1 Then Exit Sub

    Dim indiceCas As Long
    indiceCas = Target.Row - LIGNE_PREMIERE_DONNEE + 1   ' correspond à l'index dans g_CasChoix()

    ' On désactive temporairement les événements pendant nos propres écritures
    ' (sinon l'écriture dans la cellule Statut redéclencherait Worksheet_Change
    ' en boucle, précaution standard en VBA événementiel).
    Application.EnableEvents = False

    If Trim(Target.value) = "" Then
        ' La catégorie a été effacée (ex. : via le bouton "Réinitialiser cette ligne")
        g_CasChoix(indiceCas) = ""
        Me.Range(COL_STATUT & Target.Row).value = "?"
        Me.Range(COL_STATUT & Target.Row).Font.Color = RGB(150, 150, 150)
        Me.Range(COL_STATUT & Target.Row).Font.Bold = False
    Else
        ' Une catégorie vient d'être choisie dans la liste déroulante
        g_CasChoix(indiceCas) = Target.value
        Me.Range(COL_STATUT & Target.Row).value = ChrW(&H2713)   ' caractère "check" (V)
        Me.Range(COL_STATUT & Target.Row).Font.Color = RGB(30, 130, 76)
        Me.Range(COL_STATUT & Target.Row).Font.Bold = True
    End If

    Call MettreAJourCompteur(Me)

    Application.EnableEvents = True

End Sub


Private Sub Worksheet_Deactivate()

    ' Verrou "modal" : tant que g_SaisieEnCours vaut True, on empêche l'utilisateur
    ' de quitter cette feuille en le ramenant immédiatement dessus.
    If g_SaisieEnCours Then
        Me.Activate
    End If

End Sub
